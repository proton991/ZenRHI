cmake_minimum_required(VERSION 3.24)
project(ZenRHITriangle LANGUAGES CXX)
if(NOT TARGET ZenRHI::ZenRHI)
    find_package(ZenRHI 0.2 CONFIG REQUIRED)
endif()
if(NOT TARGET SDL3::SDL3)
    find_package(SDL3 CONFIG QUIET)
    if(NOT TARGET SDL3::SDL3)
        include(FetchContent)
        set(SDL_TEST_LIBRARY OFF)
        FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.zip
            URL_HASH SHA256=9cd42377704398796071b8597cd7e21da254a43bad98c4199739647adc13fa6f)
        FetchContent_MakeAvailable(SDL3)
    endif()
endif()
find_package(Vulkan REQUIRED)
find_program(GLSL_VALIDATOR glslangValidator REQUIRED HINTS "$ENV{VULKAN_SDK}/Bin")
set(shader_dir "${CMAKE_CURRENT_BINARY_DIR}/shaders")
foreach(stage IN ITEMS vert frag)
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/triangle.${stage}")
    set(output "${shader_dir}/triangle.${stage}.spv")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${shader_dir}"
        COMMAND "${GLSL_VALIDATOR}" -V "${source}" -o "${output}"
        DEPENDS "${source}" VERBATIM)
    list(APPEND shaders "${output}")
endforeach()
add_executable(ZenRHITriangle Triangle.cpp ${shaders})
target_compile_definitions(ZenRHITriangle PRIVATE TRIANGLE_SHADER_DIR="${shader_dir}/")
target_link_libraries(ZenRHITriangle PRIVATE ZenRHI::ZenRHI SDL3::SDL3 Vulkan::Headers)
if(WIN32 AND TARGET SDL3::SDL3-shared)
    add_custom_command(TARGET ZenRHITriangle POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:ZenRHITriangle> VERBATIM)
endif()
