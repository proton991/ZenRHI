cmake_minimum_required(VERSION 3.24)
project(InstalledRhiConsumer LANGUAGES CXX)
find_package(ZenRHI CONFIG REQUIRED COMPONENTS Core Vulkan ZenRHI)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ZenRHI::ZenRHI)
add_subdirectory("${EXAMPLES}/Compute" Compute)
add_subdirectory("${EXAMPLES}/Triangle" Triangle)
enable_testing()
add_test(NAME InstalledTriangle COMMAND ZenRHITriangle)
add_test(NAME InstalledStartupRetry COMMAND consumer --retry)
if(WIN32)
    add_test(NAME InstalledUnsupportedProfile COMMAND consumer --unsupported-profile)
    set_tests_properties(InstalledUnsupportedProfile PROPERTIES ENVIRONMENT "VK_INSTANCE_LAYERS=VK_LAYER_KHRONOS_profiles")
endif()
foreach(example IN ITEMS ComputeReadback AsyncCompute BindlessTextures RayQuery)
    add_test(NAME Installed${example} COMMAND ZenRHI${example})
    set_tests_properties(Installed${example} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
